// Read-only structural and custody checks. Does not invoke LexLean or Lean.
import fs from 'node:fs';import path from 'node:path';import crypto from 'node:crypto';import assert from 'node:assert/strict';
const root=path.dirname(new URL(import.meta.url).pathname),read=p=>fs.readFileSync(p),sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const p=JSON.parse(read(root+'/proposal.json')),parse=f=>JSON.parse(read(f).toString().match(/\\semanticdata\{(.*)\}/)[1]);
const checks=[];function check(name,fn){fn();checks.push(name);}
function pin(f){const b=read(f.source??f.path);assert.equal(sha(b),f.sha256);if(f.bytes!==undefined)assert.equal(b.length,f.bytes);}
check('all 15 shared input bytes unchanged',()=>{assert.equal(p.sharedInputs.length,15);p.sharedInputs.forEach(pin);});
check('prior manifest, leaf and SPEC review exact',()=>[p.priorCandidateManifest,p.leafCertificate,p.specReview].forEach(pin));
const spec=JSON.parse(read(p.specReview.path));check('proposal and official source/prior native review exact',()=>[spec.proposal,...spec.additionalPins].forEach(pin));
const leaf=parse(p.leafCertificate.path).declarations[0],context=parse(root+'/tests/lexlean/Object257StagingContext.lex.tex'),positive=parse(root+'/tests/lexlean/Object257StagingTransport.lex.tex'),negative=parse(root+'/tests/lexlean/Object257StagingTransportFalseBase.lex.tex');
const [H,P]=context.declarations,[decoy,lifted,consumer]=positive.declarations;
check('semantic-module/2, exact declaration census',()=>{for(const x of [context,positive,negative])assert.equal(x.spec,'lexlean/semantic-module/2');assert.equal(context.declarations.length,2);assert.equal(positive.declarations.length,3);assert.equal(negative.declarations.length,3);});
check('accepted actual leaf statement and Reflexivity unchanged',()=>{assert.equal(sha(JSON.stringify(leaf.statement)),p.leafCertificate.statementJsonSha256);assert.deepEqual(leaf.proof,{kind:'reflexivity'});assert.deepEqual(leaf.axioms,[]);assert.deepEqual(P.body.branches[0].body,leaf.statement);});
check('H actual staging operation and exact error-preserving output type',()=>{assert.equal(H.name,'ObjLeafStagingContext');assert.deepEqual(H.result,{kind:'result',ok:{kind:'list',element:{arguments:[],kind:'named',member:{module:'R4.ObjectTypes',name:'ObjRecord'}}},error:{arguments:[],kind:'named',member:{module:'R4.ObjectTypes',name:'ObjError'}}});const [e,o]=H.body.branches;assert.deepEqual(e.body.arguments,[{kind:'var',name:'failure'}]);assert.equal(e.body.constructor.name,'Result.error');assert.deepEqual(e.body.type_arguments,[H.result.ok,H.result.error]);assert.deepEqual(o.body,{arguments:[{kind:'var',name:'inventory'},{kind:'var',name:'leaf'},{kind:'var',name:'limits'}],function:{module:'R4.ObjectStore',name:'ObjStageRecord'},kind:'call'});});
check('P zero original equation; successor exact H equation',()=>{assert.deepEqual(P.result,{kind:'prop'});assert.equal(P.body.scrutinee.name,'count');const [z,s]=P.body.branches;assert.equal(z.constructor.name,'Nat.zero');assert.equal(s.constructor.name,'Nat.succ');assert.deepEqual(s.binders,['prior']);for(const [side,term]of [['left',leaf.statement.left],['right',leaf.statement.right]])assert.deepEqual(s.body[side],{arguments:[{kind:'var',name:'inventory'},{kind:'var',name:'limits'},term],function:{name:H.name},kind:'call'});});
check('exact Induction/Cases/Congruence/Apply proof shape and empty axioms',()=>{for(const d of [...positive.declarations,...negative.declarations])assert.deepEqual(d.axioms,[]);assert.equal(lifted.proof.kind,'induction');assert.equal(lifted.proof.scrutinee,'count');assert.deepEqual(lifted.proof.generalizing,[]);const[z,s]=lifted.proof.branches;assert.deepEqual(z,{constructor:'zero',binders:[],proof:{kind:'apply',theorem:{module:'Object257LeafCertificate',name:leaf.name},type_arguments:[],arguments:[]}});assert.deepEqual(s,{constructor:'succ',binders:['prior','ih'],proof:{kind:'cases',scrutinee:'prior',branches:[{constructor:'zero',binders:[],proof:{kind:'congruence'}},{constructor:'succ',binders:['earlier'],proof:{kind:'congruence'}}]}});assert.deepEqual(consumer.proof,{kind:'apply',theorem:{name:lifted.name},type_arguments:[],arguments:[{kind:'var',name:'inventory'},{kind:'var',name:'limits'},{kind:'nat',value:'1'}]});});
check('consumer directly exposes production staging expression',()=>{const expected=structuredClone(H.body);expected.scrutinee=leaf.statement.left;assert.deepEqual(consumer.statement.left,expected);assert.deepEqual(consumer.statement.right,{kind:'call',function:{module:'R4.ObjectStore',name:'ObjStageRecord'},arguments:[{kind:'var',name:'inventory'},leaf.statement.right.arguments[0],{kind:'var',name:'limits'}]});});
check('negative differs at only zero Apply target',()=>{const altered=structuredClone(positive);altered.declarations[1].proof.branches[0].proof.theorem={name:decoy.name};assert.deepEqual(negative,altered);assert.deepEqual(decoy.statement,{kind:'eq',left:{kind:'nat',value:'0'},right:{kind:'nat',value:'0'}});assert.deepEqual(decoy.parameters,[]);assert.deepEqual(decoy.proof,{kind:'reflexivity'});});
check('all explicit external members directly imported',()=>{for(const name of ['Object257StagingContext','Object257StagingTransport','Object257StagingTransportFalseBase']){const text=read(root+'/tests/lexlean/'+name+'.lex.tex').toString(),imports=[...text.matchAll(/\\importmodule\{([^}]+)\}/g)].map(m=>m[1]);function walk(o){if(!o||typeof o!=='object')return;if(typeof o.module==='string')assert.ok(imports.includes(o.module),o.module);Object.values(o).forEach(walk);}walk(parse(root+'/tests/lexlean/'+name+'.lex.tex'));}});
check('phase configs change entrypoint only; 300000 ms retained',()=>{const base=read(p.sharedInputs.find(f=>f.path==='lexlean.toml').source).toString();assert.ok(base.includes('300000'));for(const phase of p.phasePlan){assert.equal(phase.status,'NOT_RUN');assert.equal(read(root+'/'+phase.config).toString(),base.replace(/^entrypoints = .*$/m,'entrypoints = ["'+phase.root+'"]'));}});
console.log(JSON.stringify({status:'PASS_READ_ONLY_STRUCTURE_AND_PINS_ONLY',checks,nativeCheck:'NOT_RUN',nativeNegative:'NOT_RUN',nativePositive:'NOT_RUN',axiomReplay:'NOT_RUN',note:'No compiler invocation. Structural comparison does not establish schema admission or congruence closure.'},null,2));
